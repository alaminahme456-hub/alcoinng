    90→    90→// Local error boundary — prevents app-level crash from showing error.tsx
    91→    91→class MarketErrorBoundary extends Component<
    92→    92→  { children: ReactNode },
    93→    93→  { hasError: boolean; error: Error | null }
    94→    94→> {
    95→    95→  constructor(props: { children: ReactNode }) {
    96→    96→    super(props);
    97→    97→    this.state = { hasError: false, error: null };
    98→    98→  }
    99→    99→  static getDerivedStateFromError(error: Error) {
   100→   100→    return { hasError: true, error };
   101→   101→  }
   102→   102→  componentDidCatch(error: Error, info: ErrorInfo) {
   103→   103→    console.error('MarketView error boundary caught:', error, info);
   104→   104→  }
   105→   105→  render() {
   106→   106→    if (this.state.hasError) {
   107→   107→      return (
   108→   108→        <div className="min-h-screen flex flex-col items-center justify-center bg-background text-foreground p-4">
   109→   109→          <div className="max-w-md w-full text-center space-y-4">
   110→   110→            <div className="w-16 h-16 rounded-full bg-destructive/10 flex items-center justify-center mx-auto">
   111→   111→              <XCircle className="w-8 h-8 text-destructive" />
   112→   112→            </div>
   113→   113→            <h2 className="text-lg font-bold">Trade Error</h2>
   114→   114→            <p className="text-sm text-muted-foreground">
   115→   115→              {this.state.error?.message || 'Something went wrong with the market.'}
   116→   116→            </p>
   117→   117→            <Button
   118→   118→              onClick={() => { this.setState({ hasError: false, error: null }); }}
   119→   119→              className="gradient-gold text-gold-foreground font-semibold"
   120→   120→            >
   121→   121→              Try Again
   122→   122→            </Button>
   123→   123→          </div>
   124→   124→        </div>
   125→   125→      );
   126→   126→    }
   127→   127→    return this.props.children;
   128→   128→  }
   129→   129→}

   437→   437→    }
   438→   438→    return null;
   439→   439→  };

   443→   443→    : 0;

   445→   445→
   446→   446→  return (
