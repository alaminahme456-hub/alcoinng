    91→    91→// Local error boundary — prevents app-level crash from showing error.tsx
    92→    92→class MarketErrorBoundary extends Component<
    93→    93→  { children: ReactNode },
    94→    94→  { hasError: boolean; error: Error | null }
    95→    95→> {
    96→    96→  constructor(props: { children: ReactNode }) {
    97→    97→    super(props);
    98→    98→    this.state = { hasError: false, error: null };
    99→    99→  }
   100→   100→  static getDerivedStateFromError(error: Error) {
   101→   101→    return { hasError: true, error };
   102→   102→  }
   103→   103→  componentDidCatch(error: Error, info: ErrorInfo) {
   104→   104→    console.error('MarketView error boundary caught:', error, info);
   105→   105→  }
   106→   106→  render() {
   107→   107→    if (this.state.hasError) {
   108→   108→      return (
   109→   109→        <div className="min-h-screen flex flex-col items-center justify-center bg-background text-foreground p-4">
   110→   110→          <div className="max-w-md w-full text-center space-y-4">
   111→   111→            <div className="w-16 h-16 rounded-full bg-destructive/10 flex items-center justify-center mx-auto">
   112→   112→              <XCircle className="w-8 h-8 text-destructive" />
   113→   113→            </div>
   114→   114→            <h2 className="text-lg font-bold">Trade Error</h2>
   115→   115→            <p className="text-sm text-muted-foreground">
   116→   116→              {this.state.error?.message || 'Something went wrong with the market.'}
   117→   117→            </p>
   118→   118→            <Button
   119→   119→              onClick={() => { this.setState({ hasError: false, error: null }); }}
   120→   120→              className="gradient-gold text-gold-foreground font-semibold"
   121→   121→            >
   122→   122→              Try Again
   123→   123→            </Button>
   124→   124→          </div>
   125→   125→        </div>
   126→   126→      );
   127→   127→    }
   128→   128→    return this.props.children;
   129→   129→  }
   130→   130→}

   153→   153→    momentum: 0,
   154→   154→    volatility: 0.5,
   155→   155→    trend: 0,

   186→   186→  // Generate AI-driven price tick with momentum, trend, mean-reversion & volatility clustering
   187→   187→  const generatePriceTick = useCallback(() => {
   188→   188→    const sim = priceSimRef.current;
   189→   189→    sim.tickCount++;
   190→   190→
   191→   191→    // Shift trend slowly every ~20 ticks
   192→   192→    if (sim.tickCount % 20 === 0) {
   193→   193→      sim.trend = (Math.random() - 0.5) * 0.6;
   194→   194→    }
   195→   195→
   196→   196→    // GARCH-like volatility clustering
   197→   197→    const volTarget = 0.4 + Math.random() * 0.8;
   198→   198→    sim.volatility = 0.85 * sim.volatility + 0.15 * volTarget;
   199→   199→
   200→   200→    // Mean-reversion pull toward 75-85 range
   201→   201→    const meanTarget = 80;
   202→   202→    const meanPull = (meanTarget - sim.price) * 0.002;
   203→   203→
   204→   204→    // Random shock + momentum + trend + mean reversion
   205→   205→    const shock = (Math.random() - 0.5) * 2 * sim.volatility;
   206→   206→    sim.momentum = 0.7 * sim.momentum + 0.3 * (shock + sim.trend);
   207→   207→
   208→   208→    const delta = sim.momentum + meanPull;
   209→   209→    sim.price = Math.max(10, Math.min(200, sim.price + delta));
   210→   210→
   211→   211→    return Math.round(sim.price * 100) / 100;

   252→   252→        amount: Number(t.amount),

   274→   274→        setWallets({ reward: getBal(data.reward), deposit: getBal(data.deposit), profit: getBal(data.profit) });

   278→   278→    }

   283→   283→    initChart();

   286→   286→
   287→   287→  // Tick new price every 1.5 seconds

   289→   289→    priceIntervalRef.current = setInterval(() => {
   290→   290→      const newPrice = generatePriceTick();
   291→   291→      const timeStr = new Date().toLocaleTimeString('en-US', {
   292→   292→        hour12: false,
   293→   293→        hour: '2-digit',
   294→   294→        minute: '2-digit',
   295→   295→        second: '2-digit',

   298→   298→      setCurrentPrice(newPrice);

   301→   301→        return next.slice(-60);
   302→   302→      });
   303→   303→    }, 1500);
   304→   304→    return () => {
   305→   305→      if (priceIntervalRef.current) clearInterval(priceIntervalRef.current);
   306→   306→    };

   308→   308→
   309→   309→  // Derived values (must be before callbacks that reference them)
   310→   310→  const availableBalance = wallets[WALLET_CONFIG[selectedWallet].balanceKey];
   311→   311→  const amountNum = parseFloat(tradeAmount) || 0;
   312→   312→
   313→   313→  // Trade countdown — ref-based, avoids all stale closure issues
   314→   314→  const countdownRef = useRef<ReturnType<typeof setInterval> | null>(null);
   315→   315→  const countdownDisplayRef = useRef(0);
   316→   316→  const pendingTradeResult = useRef<any>(null);
   317→   317→  const [, forceUpdate] = useState(0);
   318→   318→  const tradeStartTimeRef = useRef<number>(0);
   319→   319→
   320→   320→  const startCountdown = useCallback((seconds: number) => {
   321→   321→    if (countdownRef.current) {
   322→   322→      clearInterval(countdownRef.current);
   323→   323→      countdownRef.current = null;
   324→   324→    }

   326→   326→

   328→   328→      countdownDisplayRef.current -= 1;
   329→   329→      forceUpdate((n) => n + 1);

   331→   331→        if (countdownRef.current) {
   332→   332→          clearInterval(countdownRef.current);
   333→   333→          countdownRef.current = null;
   334→   334→        }

   408→   408→      const totalReturn = isWin ? Math.round(amountNum * multiplierUsed * 100) / 100 : 0;
   409→   409→      const profit = isWin ? Math.round((totalReturn - amountNum) * 100) / 100 : 0;
   410→   410→
   411→   411→      const precomputed = {
   412→   412→        win: isWin,
   413→   413→        profit: Number(profit) || 0,
   414→   414→        totalReturn: Number(totalReturn) || 0,
   415→   415→        multiplier: Number(multiplierUsed) || 1,
   416→   416→        message: data.message || (isWin ? 'Trade successful!' : 'Better luck next time.'),
   417→   417→      };
   418→   418→
   419→   419→      // Store monitor info
   420→   420→      const now = Date.now();
   421→   421→      const startPrice = Number(data.trade?.start_price) || currentPrice || 80;
   422→   422→      monitorInfoRef.current = {
   423→   423→        tradeId: data.trade?.id || `AL${Date.now().toString(36).toUpperCase()}`,
   424→   424→        prediction: prediction!,
   425→   425→        startPrice,
   426→   426→        startTime: now,
   427→   427→        endTime: now + duration * 1000,
   428→   428→        duration,
   429→   429→        amount: amountNum,
   430→   430→      };
   431→   431→      tradeStartTimeRef.current = now;
   432→   432→
   433→   433→      // If countdown already finished (very fast), show immediately
   434→   434→      if (countdownDisplayRef.current <= 0) {
   435→   435→        setTradeResult(precomputed);
   436→   436→        setTradePhase('result');
   437→   437→        toast.success(isWin ? `Trade Won! +${formatNaira(profit)}` : 'Trade Lost');
   438→   438→        setTrading(false);
   439→   439→        refreshWallets();
   440→   440→        fetchHistory();
   441→   441→      } else {
   442→   442→        // Start countdown, transition confirmed → monitoring after 1.5s
   443→   443→        startCountdown(duration);
   444→   444→        setTimeout(() => {
   445→   445→          if (trading) setTradePhase('monitoring');
   446→   446→        }, 1500);
   447→   447→        pendingTradeResult.current = precomputed;
   448→   448→      }
   449→   449→    } catch (err: any) {
   450→   450→      stopCountdown();
   451→   451→      pendingTradeResult.current = null;
   452→   452→      monitorInfoRef.current = null;
   453→   453→      setTrading(false);
   454→   454→      setTradePhase('idle');
   455→   455→      toast.error(err.message || 'Trade failed');
   456→   456→    }
   457→   457→  };
   458→   458→
   459→   459→  const filteredHistory = historyFilter === 'all'
   460→   460→    ? history
   461→   461→    : history.filter((t) => t.result === historyFilter);
   462→   462→
   463→   463→  // Custom chart tooltip

   469→   469→          <p className="font-semibold text-gold">{formatNaira(payload[0].value)}</p>
   470→   470→        </div>
   471→   471→      );
   472→   472→    }
   473→   473→    return null;
