    91→    91→// Local error boundary — prevents app-level crash from showing error.tsx
    92→    92→class MarketErrorBoundary extends Component<
    93→    93→  { children: ReactNode },
    94→    94→  { hasError: boolean; error: Error | null }
    95→    95→> {
    96→    96→  constructor(props: { children: ReactNode }) {
    97→    97→    super(props);
    98→    98→    this.state = { hasError: false, error: null };
    99→    99→  }
   100→   100→  static getDerivedStateFromError(error: Error) {
   101→   101→    return { hasError: true, error };
   102→   102→  }
   103→   103→  componentDidCatch(error: Error, info: ErrorInfo) {
   104→   104→    console.error('MarketView error boundary caught:', error, info);
   105→   105→  }
   106→   106→  render() {
   107→   107→    if (this.state.hasError) {
   108→   108→      return (
   109→   109→        <div className="min-h-screen flex flex-col items-center justify-center bg-background text-foreground p-4">
   110→   110→          <div className="max-w-md w-full text-center space-y-4">
   111→   111→            <div className="w-16 h-16 rounded-full bg-destructive/10 flex items-center justify-center mx-auto">
   112→   112→              <XCircle className="w-8 h-8 text-destructive" />
   113→   113→            </div>
   114→   114→            <h2 className="text-lg font-bold">Trade Error</h2>
   115→   115→            <p className="text-sm text-muted-foreground">
   116→   116→              {this.state.error?.message || 'Something went wrong with the market.'}
   117→   117→            </p>
   118→   118→            <Button
   119→   119→              onClick={() => { this.setState({ hasError: false, error: null }); }}
   120→   120→              className="gradient-gold text-gold-foreground font-semibold"
   121→   121→            >
   122→   122→              Try Again
   123→   123→            </Button>
   124→   124→          </div>
   125→   125→        </div>
   126→   126→      );
   127→   127→    }
   128→   128→    return this.props.children;
   129→   129→  }
   130→   130→}

   462→   462→        }, 1500);
   463→   463→        pendingTradeResult.current = precomputed;
   464→   464→      }
   465→   465→    } catch (err: any) {

   468→   468→      monitorInfoRef.current = null;
   469→   469→      setTrading(false);
