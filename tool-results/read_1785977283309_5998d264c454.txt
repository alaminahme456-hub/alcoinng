     2→

     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import { useAppStore, apiFetch } from '@/store';
     6→import { Button } from '@/components/ui/button';
     7→import { Input } from '@/components/ui/input';
     8→import { Badge } from '@/components/ui/badge';
     9→import { Skeleton } from '@/components/ui/skeleton';
    10→import {
    11→  Select,
    12→  SelectContent,
    13→  SelectItem,
    14→  SelectTrigger,
    15→  SelectValue,
    16→} from '@/components/ui/select';
    17→import {
    18→  Table,
    19→  TableBody,
    20→  TableCell,
    21→  TableHead,
    22→  TableHeader,
    23→  TableRow,
    24→} from '@/components/ui/table';
    25→import { toast } from 'sonner';
    26→import {
    27→  ArrowLeft, TrendingUp, Wallet, PiggyBank, TrendingUpIcon,
    28→  ArrowUpCircle, ArrowDownCircle, Loader2, AlertCircle,
    29→  Trophy, XCircle, CheckCircle2, Clock, History, Filter,
    30→} from 'lucide-react';

    32→  LineChart,
    33→  Line,
    34→  XAxis,
    35→  YAxis,
    36→  CartesianGrid,
    37→  Tooltip,
    38→  ResponsiveContainer,
    39→  ReferenceLine,
    40→} from 'recharts';
    41→
    42→function formatNaira(amount: number) {
    43→  return `\u20a6${amount.toLocaleString()}`;
    44→}
    45→
    46→interface PricePoint {
    47→  time: string;
    48→  price: number;
    49→  timestamp: number;
    50→}
    51→
    52→interface TradeHistoryItem {
    53→  id: string;
    54→  prediction: 'UP' | 'DOWN';
    55→  amount: number;
    56→  multiplier: number;
    57→  duration: number;
    58→  result: 'win' | 'loss';
    59→  profit: number;
    60→  createdAt: string;
    61→  entryPrice: number;
    62→  exitPrice: number;
    63→}
    64→
    65→type WalletType = 'reward' | 'deposit' | 'profit';
    66→type Prediction = 'UP' | 'DOWN' | null;
    67→
    68→const DURATION_OPTIONS = [
    69→  { label: '10s', value: 10 },
    70→  { label: '30s', value: 30 },
    71→  { label: '1min', value: 60 },
    72→  { label: '2min', value: 120 },
    73→  { label: '5min', value: 300 },
    74→  { label: '10min', value: 600 },
    75→];
    76→
    77→const MULTIPLIER_OPTIONS = [
    78→  { label: '1.10x', value: 1.10 },
    79→  { label: '1.20x', value: 1.20 },
    80→  { label: '1.30x', value: 1.30 },
    81→  { label: '1.40x', value: 1.40 },
    82→  { label: '1.50x', value: 1.50 },
    83→];
    84→
    85→const WALLET_CONFIG: Record<WalletType, { label: string; icon: typeof Wallet; color: string; balanceKey: 'reward' | 'deposit' | 'profit' }> = {
    86→  reward: { label: 'Reward', icon: Wallet, color: 'text-gold', balanceKey: 'reward' },
    87→  deposit: { label: 'Deposit', icon: PiggyBank, color: 'text-alcoin-blue', balanceKey: 'deposit' },
    88→  profit: { label: 'Profit', icon: TrendingUpIcon, color: 'text-emerald-400', balanceKey: 'profit' },
    89→};
    90→
    91→const HISTORY_FILTERS = [
    92→  { label: 'All', value: 'all' },
    93→  { label: 'Wins', value: 'win' },
    94→  { label: 'Losses', value: 'loss' },
    95→];
    96→
    97→export default function MarketView() {
    98→  const { wallets, setWallets, setView } = useAppStore();
    99→
   100→  // Chart data
   101→  const [priceData, setPriceData] = useState<PricePoint[]>([]);
   102→  const [currentPrice, setCurrentPrice] = useState<number | null>(null);
   103→  const [prevPrice, setPrevPrice] = useState<number | null>(null);
   104→  const [chartLoading, setChartLoading] = useState(true);
   105→  const priceIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   106→
   107→  // Trade form
   108→  const [selectedWallet, setSelectedWallet] = useState<WalletType>('deposit');
   109→  const [prediction, setPrediction] = useState<Prediction>(null);
   110→  const [tradeAmount, setTradeAmount] = useState('');
   111→  const [duration, setDuration] = useState<number>(30);
   112→  const [multiplier, setMultiplier] = useState<number>(1.20);
   113→
   114→  // Trade execution
   115→  const [trading, setTrading] = useState(false);
   116→  const [tradeCountdown, setTradeCountdown] = useState<number | null>(null);
   117→  const [tradeResult, setTradeResult] = useState<{ win: boolean; profit: number; message: string } | null>(null);
   118→
   119→  // History
   120→  const [history, setHistory] = useState<TradeHistoryItem[]>([]);
   121→  const [historyLoading, setHistoryLoading] = useState(true);
   122→  const [historyFilter, setHistoryFilter] = useState('all');
   123→
   124→  const fetchPrice = useCallback(async () => {
   125→    try {
   126→      const data = await apiFetch('/api/market/price');
   127→      if (data.price !== undefined) {
   128→        setPrevPrice(currentPrice);
   129→        setCurrentPrice(data.price);
   130→        if (data.history && Array.isArray(data.history)) {
   131→          setPriceData(data.history.slice(-60));
   132→        } else if (data.time) {
   133→          setPriceData((prev) => {
   134→            const next = [...prev, { time: data.time, price: data.price, timestamp: Date.now() }];
   135→            return next.slice(-60);
   136→          });
   137→        }
   138→      }
   139→      setChartLoading(false);
   140→    } catch {
   141→      if (chartLoading) setChartLoading(false);
   142→    }
   143→  }, [currentPrice, chartLoading]);
   144→
   145→  const fetchHistory = useCallback(async () => {
   146→    try {
   147→      setHistoryLoading(true);
   148→      const data = await apiFetch('/api/market/history');
   149→      setHistory(data.trades || []);
   150→    } catch {
   151→      // silent
   152→    } finally {
   153→      setHistoryLoading(false);
   154→    }
   155→  }, []);
   156→
   157→  const refreshWallets = useCallback(async () => {
   158→    try {
   159→      const data = await apiFetch('/api/user/wallets');
   160→      if (data) {
   161→        const getBal = (w: any) => typeof w === 'object' && w !== null ? (w.balance ?? 0) : (w ?? 0);
   162→        setWallets({ reward: getBal(data.reward), deposit: getBal(data.deposit), profit: getBal(data.profit) });
   163→      }
   164→    } catch {
   165→      // silent
   166→    }
   167→  }, [setWallets]);
   168→
   169→  // Initial fetch
   170→  useEffect(() => {
   171→    fetchPrice();
   172→    fetchHistory();
   173→  }, [fetchPrice, fetchHistory]);
   174→
   175→  // Poll price every 2s
   176→  useEffect(() => {
   177→    priceIntervalRef.current = setInterval(fetchPrice, 2000);
   178→    return () => {
   179→      if (priceIntervalRef.current) clearInterval(priceIntervalRef.current);
   180→    };
   181→  }, [fetchPrice]);
   182→
   183→  // Trade countdown
   184→  const countdownRef = useRef<ReturnType<typeof setInterval> | null>(null);
   185→
   186→  useEffect(() => {
   187→    if (tradeCountdown === null) return;
   188→
   189→    countdownRef.current = setInterval(() => {
   190→      setTradeCountdown((prev) => {
   191→        if (prev === null || prev <= 1) {
   192→          if (countdownRef.current) clearInterval(countdownRef.current);
   193→          return 0;
   194→        }
   195→        return prev - 1;
   196→      });
   197→    }, 1000);
   198→
   199→    return () => {
   200→      if (countdownRef.current) clearInterval(countdownRef.current);
   201→    };
   202→  }, [tradeCountdown]);
   203→
   204→  const availableBalance = wallets[WALLET_CONFIG[selectedWallet].balanceKey];
   205→  const amountNum = parseFloat(tradeAmount) || 0;
   206→  const estimatedProfit = prediction && amountNum > 0
   207→    ? amountNum * multiplier - amountNum
   208→    : 0;
   209→
   210→  const formatCountdown = (seconds: number) => {
   211→    const m = Math.floor(seconds / 60);
   212→    const s = seconds % 60;
   213→    if (m > 0) return `${m}:${s.toString().padStart(2, '0')}`;
   214→    return `0:${s.toString().padStart(2, '0')}`;
   215→  };
   216→
   217→  const handleTrade = async () => {
   218→    if (!prediction || amountNum <= 0) return;
   219→    if (amountNum > availableBalance) {
   220→      toast.error('Insufficient balance', {
   221→        description: `Your ${WALLET_CONFIG[selectedWallet].label} wallet has ${formatNaira(availableBalance)}`,
   222→      });
   223→      return;
   224→    }
   225→
   226→    setTrading(true);
   227→    setTradeResult(null);
   228→    setTradeCountdown(duration);
   229→
   230→    try {
   231→      const data = await apiFetch('/api/market/trade', {
   232→        method: 'POST',
   233→        body: JSON.stringify({
   234→          wallet: selectedWallet,
   235→          prediction,
   236→          amount: amountNum,
   237→          duration,
   238→          multiplier,
   239→        }),
   240→      });
   241→
   242→      setTradeCountdown(0);
   243→
   244→      // Small delay for dramatic effect
   245→      await new Promise((r) => setTimeout(r, 400));
   246→
   247→      setTradeResult({
   248→        win: data.result === 'win',
   249→        profit: data.profit || 0,
   250→        message: data.message || (data.result === 'win' ? 'Trade successful!' : 'Better luck next time.'),
   251→      });
   252→
   253→      toast.success(
   254→        data.result === 'win'
   255→          ? `Trade Won! +${formatNaira(data.profit || 0)}`
   256→          : `Trade Lost`,
   257→        {
   258→          description: data.message || '',
   259→        }
   260→      );
   261→
   262→      refreshWallets();
   263→      fetchHistory();
   264→    } catch (err: any) {
   265→      setTradeCountdown(null);
   266→      toast.error(err.message || 'Trade failed');
   267→    } finally {
   268→      setTrading(false);
   269→    }
   270→  };
   271→
   272→  const filteredHistory = historyFilter === 'all'
   273→    ? history
   274→    : history.filter((t) => t.result === historyFilter);
   275→
   276→  // Custom chart tooltip
   277→  const CustomTooltip = ({ active, payload }: any) => {
   278→    if (active && payload && payload.length) {
   279→      return (
   280→        <div className="glass-strong rounded-lg p-2 text-xs border border-white/10">
   281→          <p className="text-muted-foreground">{payload[0].payload.time}</p>
   282→          <p className="font-semibold text-gold">{formatNaira(payload[0].value)}</p>
   283→        </div>
   284→      );
   285→    }
   286→    return null;
   287→  };
   288→
   289→  const priceChange = prevPrice !== null && currentPrice !== null
   290→    ? currentPrice - prevPrice
   291→    : 0;
   292→  const priceUp = priceChange >= 0;
   293→
   294→  return (
   295→    <div className="min-h-screen pb-8">
   296→      {/* Header */}
   297→      <header className="sticky top-0 z-30 glass-strong px-4 py-3 flex items-center gap-3">
   298→        <button
   299→          onClick={() => setView('dashboard')}
   300→          className="w-10 h-10 rounded-xl glass flex items-center justify-center hover:bg-white/10 transition-colors"
   301→        >
   302→          <ArrowLeft className="w-5 h-5" />
   303→        </button>
   304→        <div className="flex items-center gap-2">
   305→          <div className="w-8 h-8 rounded-lg gradient-gold flex items-center justify-center">
   306→            <TrendingUp className="w-4 h-4 text-gold-foreground" />
   307→          </div>
   308→          <h1 className="font-semibold text-lg">AL Coin Market</h1>
   309→        </div>
   310→      </header>
   311→
   312→      <main className="px-4 pt-4 max-w-2xl mx-auto space-y-4">
   313→        {/* Price Chart */}
   314→        <motion.div
   315→          initial={{ opacity: 0, y: 20 }}
   316→          animate={{ opacity: 1, y: 0 }}
   317→          className="glass rounded-xl p-4"
   318→        >
   319→          <div className="flex items-center justify-between mb-3">
   320→            <h2 className="text-sm font-semibold text-muted-foreground">Live Price Chart</h2>
   321→            <div className="flex items-center gap-1.5">
   322→              <div className="w-2 h-2 rounded-full bg-emerald-400 animate-pulse" />
   323→              <span className="text-[10px] text-muted-foreground">LIVE</span>
   324→            </div>
   325→          </div>
   326→
   327→          {chartLoading ? (
   328→            <Skeleton className="h-48 rounded-lg bg-white/5" />
   329→          ) : priceData.length === 0 ? (
   330→            <div className="h-48 rounded-lg bg-white/5 flex items-center justify-center">
   331→              <p className="text-sm text-muted-foreground">Waiting for price data...</p>
   332→            </div>
   333→          ) : (
   334→            <div className="h-48 sm:h-56">
   335→              <ResponsiveContainer width="100%" height="100%">
   336→                <LineChart data={priceData} margin={{ top: 5, right: 5, left: -20, bottom: 0 }}>
   337→                  <CartesianGrid
   338→                    strokeDasharray="3 3"
   339→                    stroke="rgba(212, 175, 55, 0.08)"
   340→                    vertical={false}
   341→                  />
   342→                  <XAxis
   343→                    dataKey="time"
   344→                    tick={{ fontSize: 10, fill: '#8888a0' }}
   345→                    axisLine={{ stroke: 'rgba(255,255,255,0.06)' }}
   346→                    tickLine={false}
   347→                    interval="preserveStartEnd"
   348→                  />
   349→                  <YAxis
   350→                    domain={['auto', 'auto']}
   351→                    tick={{ fontSize: 10, fill: '#8888a0' }}
   352→                    axisLine={false}
   353→                    tickLine={false}
   354→                    width={50}
   355→                    tickFormatter={(v) => `₦${v.toLocaleString()}`}
   356→                  />
   357→                  <Tooltip content={<CustomTooltip />} />
   358→                  <Line
   359→                    type="monotone"
   360→                    dataKey="price"
   361→                    stroke={priceUp ? '#10b981' : '#ef4444'}
   362→                    strokeWidth={2}
   363→                    dot={false}
   364→                    activeDot={{ r: 4, fill: priceUp ? '#10b981' : '#ef4444', strokeWidth: 0 }}
   365→                  />
   366→                  <ReferenceLine
   367→                    y={priceData.length > 0 ? priceData[priceData.length - 1]?.price : undefined}
   368→                    stroke="rgba(212, 175, 55, 0.2)"
   369→                    strokeDasharray="3 3"
   370→                  />
   371→                </LineChart>
   372→              </ResponsiveContainer>
   373→            </div>
   374→          )}
   375→        </motion.div>
   376→
   377→        {/* Current Price */}
   378→        <motion.div
   379→          initial={{ opacity: 0, y: 20 }}
   380→          animate={{ opacity: 1, y: 0 }}
   381→          transition={{ delay: 0.05 }}
   382→          className="glass rounded-xl p-4 text-center gold-glow"
   383→        >
   384→          <p className="text-xs text-muted-foreground mb-1">AL Coin Current Price</p>
   385→          <AnimatePresence mode="wait">
   386→            <motion.p
   387→              key={currentPrice}
   388→              initial={{ opacity: 0, y: 5 }}
   389→              animate={{ opacity: 1, y: 0 }}
   390→              exit={{ opacity: 0, y: -5 }}
   391→              className={`text-3xl sm:text-4xl font-bold font-mono ${
   392→                priceUp ? 'text-emerald-400' : priceChange !== 0 ? 'text-red-400' : 'gradient-gold-text'
   393→              }`}
   394→            >
   395→              {currentPrice !== null ? formatNaira(currentPrice) : '---'}
   396→            </motion.p>
   397→          </AnimatePresence>
   398→          {priceChange !== 0 && (
   399→            <p className={`text-xs mt-1 font-medium ${priceUp ? 'text-emerald-400' : 'text-red-400'}`}>
   400→              {priceUp ? '▲' : '▼'} {priceUp ? '+' : ''}{formatNaira(Math.abs(priceChange))}
   401→            </p>
   402→          )}
   403→        </motion.div>
   404→
   405→        {/* Trade Form */}
   406→        <motion.div
   407→          initial={{ opacity: 0, y: 20 }}
   408→          animate={{ opacity: 1, y: 0 }}
   409→          transition={{ delay: 0.1 }}
   410→          className="glass rounded-xl p-4 space-y-4"
   411→        >
   412→          <h2 className="font-semibold text-sm">Place Trade</h2>
   413→
   414→          {/* Wallet Selector */}
   415→          <div className="space-y-2">
   416→            <p className="text-xs text-muted-foreground">Select Wallet</p>
   417→            <div className="grid grid-cols-3 gap-2">
   418→              {(Object.keys(WALLET_CONFIG) as WalletType[]).map((key) => {
   419→                const cfg = WALLET_CONFIG[key];
   420→                const isSelected = selectedWallet === key;
   421→                const bal = wallets[cfg.balanceKey];
   422→
   423→                return (
   424→                  <button
   425→                    key={key}
   426→                    onClick={() => setSelectedWallet(key)}
   427→                    className={`rounded-xl p-3 text-center transition-all border ${
   428→                      isSelected
   429→                        ? 'glass-strong border-gold/40 gold-glow'
   430→                        : 'glass border-transparent hover:border-white/10'
   431→                    }`}
   432→                  >
   433→                    <cfg.icon className={`w-4 h-4 mx-auto mb-1 ${cfg.color}`} />
   434→                    <p className={`text-xs font-medium ${isSelected ? cfg.color : 'text-muted-foreground'}`}>
   435→                      {cfg.label}
   436→                    </p>
   437→                    <p className={`text-xs font-bold mt-0.5 ${isSelected ? cfg.color : 'text-foreground'}`}>
   438→                      {formatNaira(bal)}
   439→                    </p>
   440→                  </button>
   441→                );
   442→              })}
   443→            </div>
   444→          </div>
   445→
   446→          {/* Prediction */}
   447→          <div className="space-y-2">
   448→            <p className="text-xs text-muted-foreground">Prediction</p>
   449→            <div className="grid grid-cols-2 gap-3">
   450→              <motion.button
   451→                whileTap={{ scale: 0.97 }}
   452→                onClick={() => setPrediction(prediction === 'UP' ? null : 'UP')}
   453→                className={`rounded-xl p-4 flex flex-col items-center gap-2 transition-all border-2 min-h-[72px] ${
   454→                  prediction === 'UP'
   455→                    ? 'bg-emerald-500/15 border-emerald-500/50'
   456→                    : 'glass border-transparent hover:border-emerald-500/20'
   457→                }`}
   458→              >
   459→                <ArrowUpCircle className={`w-7 h-7 ${prediction === 'UP' ? 'text-emerald-400' : 'text-muted-foreground'}`} />
   460→                <span className={`text-sm font-bold ${prediction === 'UP' ? 'text-emerald-400' : 'text-muted-foreground'}`}>
   461→                  BUY (UP)
   462→                </span>
   463→              </motion.button>
   464→
   465→              <motion.button
   466→                whileTap={{ scale: 0.97 }}
   467→                onClick={() => setPrediction(prediction === 'DOWN' ? null : 'DOWN')}
   468→                className={`rounded-xl p-4 flex flex-col items-center gap-2 transition-all border-2 min-h-[72px] ${
   469→                  prediction === 'DOWN'
   470→                    ? 'bg-red-500/15 border-red-500/50'
   471→                    : 'glass border-transparent hover:border-red-500/20'
   472→                }`}
   473→              >
   474→                <ArrowDownCircle className={`w-7 h-7 ${prediction === 'DOWN' ? 'text-red-400' : 'text-muted-foreground'}`} />
   475→                <span className={`text-sm font-bold ${prediction === 'DOWN' ? 'text-red-400' : 'text-muted-foreground'}`}>
   476→                  SELL (DOWN)
   477→                </span>
   478→              </motion.button>
   479→            </div>
   480→          </div>
   481→
   482→          {/* Investment Amount */}
   483→          <div className="space-y-2">
   484→            <p className="text-xs text-muted-foreground">Investment Amount</p>
   485→            <div className="relative">
   486→              <span className="absolute left-3 top-1/2 -translate-y-1/2 text-sm text-muted-foreground font-medium">
   487→                ₦
   488→              </span>
   489→              <Input
   490→                type="number"
   491→                inputMode="decimal"
   492→                placeholder="0"
   493→                value={tradeAmount}
   494→                onChange={(e) => setTradeAmount(e.target.value)}
   495→                className="pl-8 pr-16 bg-white/5 border-white/10 focus:border-gold h-12 text-lg font-semibold"
   496→                min="0"
   497→                max={availableBalance}
   498→              />
   499→              <button
   500→                type="button"
   501→                onClick={() => setTradeAmount(availableBalance.toString())}
   502→                className="absolute right-2 top-1/2 -translate-y-1/2 px-2 py-1 rounded text-[10px] font-medium text-gold bg-gold/10 hover:bg-gold/20 transition-colors"
   503→              >
   504→                MAX
   505→              </button>
   506→            </div>
   507→            {/* Quick amounts */}
   508→            <div className="flex gap-2">
   509→              {[100, 500, 1000, 2000, 5000].map((amt) => (
   510→                <button
   511→                  key={amt}
   512→                  onClick={() => setTradeAmount(amt.toString())}
   513→                  className="flex-1 glass rounded-lg py-1.5 text-[10px] font-medium text-muted-foreground hover:text-foreground hover:bg-white/8 transition-colors"
   514→                >
   515→                  {amt >= 1000 ? `${amt / 1000}K` : amt}
   516→                </button>
   517→              ))}
   518→            </div>
   519→          </div>
   520→
   521→          {/* Duration & Multiplier Row */}
   522→          <div className="grid grid-cols-2 gap-3">
   523→            <div className="space-y-2">
   524→              <p className="text-xs text-muted-foreground">Duration</p>
   525→              <Select
   526→                value={duration.toString()}
   527→                onValueChange={(v) => setDuration(parseInt(v))}
   528→              >
   529→                <SelectTrigger className="w-full bg-white/5 border-white/10 focus:border-gold h-11">
   530→                  <SelectValue />
   531→                </SelectTrigger>
   532→                <SelectContent className="glass-strong border-white/10">
   533→                  {DURATION_OPTIONS.map((opt) => (
   534→                    <SelectItem key={opt.value} value={opt.value.toString()}>
   535→                      {opt.label}
   536→                    </SelectItem>
   537→                  ))}
   538→                </SelectContent>
   539→              </Select>
   540→            </div>
   541→
   542→            <div className="space-y-2">
   543→              <p className="text-xs text-muted-foreground">Multiplier</p>
   544→              <Select
   545→                value={multiplier.toString()}
   546→                onValueChange={(v) => setMultiplier(parseFloat(v))}
   547→              >
   548→                <SelectTrigger className="w-full bg-white/5 border-white/10 focus:border-gold h-11">
   549→                  <SelectValue />
   550→                </SelectTrigger>
   551→                <SelectContent className="glass-strong border-white/10">

   798→            </div>
   799→          </div>
   800→
   801→          {historyLoading ? (

   803→              {Array.from({ length: 3 }).map((_, i) => (

   885→                </TableBody>
   886→              </Table>
   887→            </div>
