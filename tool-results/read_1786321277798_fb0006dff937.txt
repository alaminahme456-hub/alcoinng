    91→// Local error boundary — prevents app-level crash from showing error.tsx
    92→class MarketErrorBoundary extends Component<
    93→  { children: ReactNode },
    94→  { hasError: boolean; error: Error | null }
    95→> {
    96→  constructor(props: { children: ReactNode }) {
    97→    super(props);
    98→    this.state = { hasError: false, error: null };
    99→  }
   100→  static getDerivedStateFromError(error: Error) {
   101→    return { hasError: true, error };
   102→  }
   103→  componentDidCatch(error: Error, info: ErrorInfo) {
   104→    console.error('MarketView error boundary caught:', error, info);
   105→  }
   106→  render() {
   107→    if (this.state.hasError) {
   108→      return (
   109→        <div className="min-h-screen flex flex-col items-center justify-center bg-background text-foreground p-4">
   110→          <div className="max-w-md w-full text-center space-y-4">
   111→            <div className="w-16 h-16 rounded-full bg-destructive/10 flex items-center justify-center mx-auto">
   112→              <XCircle className="w-8 h-8 text-destructive" />
   113→            </div>
   114→            <h2 className="text-lg font-bold">Trade Error</h2>
   115→            <p className="text-sm text-muted-foreground">
   116→              {this.state.error?.message || 'Something went wrong with the market.'}
   117→            </p>
   118→            <Button
   119→              onClick={() => { this.setState({ hasError: false, error: null }); }}
   120→              className="gradient-gold text-gold-foreground font-semibold"
   121→            >
   122→              Try Again
   123→            </Button>
   124→          </div>
   125→        </div>
   126→      );
   127→    }
   128→    return this.props.children;
   129→  }
   130→}

   190→
   191→    // Shift trend slowly every ~20 ticks
   192→    if (sim.tickCount % 20 === 0) {
   193→      sim.trend = (Math.random() - 0.5) * 0.6;
   194→    }
   195→
   196→    // GARCH-like volatility clustering
   197→    const volTarget = 0.4 + Math.random() * 0.8;
   198→    sim.volatility = 0.85 * sim.volatility + 0.15 * volTarget;
   199→
   200→    // Mean-reversion pull toward 75-85 range
   201→    const meanTarget = 80;
   202→    const meanPull = (meanTarget - sim.price) * 0.002;
   203→
   204→    // Random shock + momentum + trend + mean reversion
   205→    const shock = (Math.random() - 0.5) * 2 * sim.volatility;
   206→    sim.momentum = 0.7 * sim.momentum + 0.3 * (shock + sim.trend);
   207→
   208→    const delta = sim.momentum + meanPull;
   209→    sim.price = Math.max(10, Math.min(200, sim.price + delta));

   424→        prediction: prediction!,
   425→        startPrice,
   426→        startTime: now,
   427→        endTime: now + duration * 1000,
   428→        duration,
   429→        amount: amountNum,
   430→      };
   431→      tradeStartTimeRef.current = now;
   432→
   433→      // If countdown already finished (very fast), show immediately
   434→      if (countdownDisplayRef.current <= 0) {
   435→        setTradeResult(precomputed);
   436→        setTradePhase('result');
   437→        toast.success(isWin ? `Trade Won! +${formatNaira(profit)}` : 'Trade Lost');
   438→        setTrading(false);
   439→        refreshWallets();
   440→        fetchHistory();

   445→          if (trading) setTradePhase('monitoring');
   446→        }, 1500);

   448→      }
   449→    } catch (err: any) {
   450→      stopCountdown();
   451→      pendingTradeResult.current = null;
   452→      monitorInfoRef.current = null;
   453→      setTrading(false);
   454→      setTradePhase('idle');

   474→  };

   478→    : 0;

   485→        <button

   488→        >
   489→          <ArrowLeft className="w-5 h-5" />
   490→        </button>
   491→        <div className="flex items-center gap-2">
   492→          <div className="w-8 h-8 rounded-lg gradient-gold flex items-center justify-center">
   493→            <TrendingUp className="w-4 h-4 text-gold-foreground" />

   497→      </header>

   499→      <main className="px-4 pt-4 max-w-2xl mx-auto space-y-4">
   500→        {/* Price Chart */}
   501→        <motion.div
   502→          initial={{ opacity: 0, y: 20 }}

   506→          <div className="flex items-center justify-between mb-3">

   508→            <div className="flex items-center gap-1.5">
   509→              <div className="w-2 h-2 rounded-full bg-emerald-400 animate-pulse" />

   511→            </div>

   526→                      <stop offset="0%" stopColor="#34d399" stopOpacity={0.25} />

   529→                    <linearGradient id="priceGradientDown" x1="0" y1="0" x2="0" y2="1">
