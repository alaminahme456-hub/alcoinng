    88→// Local error boundary — prevents app-level crash from showing error.tsx
    89→class MarketErrorBoundary extends Component<
    90→  { children: ReactNode },
    91→  { hasError: boolean; error: Error | null }
    92→> {
    93→  constructor(props: { children: ReactNode }) {
    94→    super(props);
    95→    this.state = { hasError: false, error: null };
    96→  }
    97→  static getDerivedStateFromError(error: Error) {
    98→    return { hasError: true, error };
    99→  }
   100→  componentDidCatch(error: Error, info: ErrorInfo) {
   101→    console.error('MarketView error boundary caught:', error, info);
   102→  }
   103→  render() {
   104→    if (this.state.hasError) {
   105→      return (
   106→        <div className="min-h-screen flex flex-col items-center justify-center bg-background text-foreground p-4">
   107→          <div className="max-w-md w-full text-center space-y-4">
   108→            <div className="w-16 h-16 rounded-full bg-destructive/10 flex items-center justify-center mx-auto">
   109→              <XCircle className="w-8 h-8 text-destructive" />
   110→            </div>
   111→            <h2 className="text-lg font-bold">Trade Error</h2>
   112→            <p className="text-sm text-muted-foreground">
   113→              {this.state.error?.message || 'Something went wrong with the market.'}
   114→            </p>
   115→            <Button
   116→              onClick={() => { this.setState({ hasError: false, error: null }); }}
   117→              className="gradient-gold text-gold-foreground font-semibold"
   118→            >
   119→              Try Again
   120→            </Button>
   121→          </div>
   122→        </div>
   123→      );
   124→    }
   125→    return this.props.children;
   126→  }
   127→}

   175→

   177→    if (sim.tickCount % 20 === 0) {
   178→      sim.trend = (Math.random() - 0.5) * 0.6;
   179→    }
   180→
   181→    // GARCH-like volatility clustering
   182→    const volTarget = 0.4 + Math.random() * 0.8;
   183→    sim.volatility = 0.85 * sim.volatility + 0.15 * volTarget;
   184→
   185→    // Mean-reversion pull toward 75-85 range
   186→    const meanTarget = 80;
   187→    const meanPull = (meanTarget - sim.price) * 0.002;
   188→
   189→    // Random shock + momentum + trend + mean reversion
   190→    const shock = (Math.random() - 0.5) * 2 * sim.volatility;
   191→    sim.momentum = 0.7 * sim.momentum + 0.3 * (shock + sim.trend);
   192→
   193→    const delta = sim.momentum + meanPull;
   194→    sim.price = Math.max(10, Math.min(200, sim.price + delta));
   195→
   196→    return Math.round(sim.price * 100) / 100;

   399→        const multiplierUsed = data.trade?.payout_multiplier || 1.0;
