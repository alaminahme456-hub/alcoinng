     1→'use client';
     2→
     3→import { useEffect, useState, useCallback, useRef } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import { useAppStore, apiFetch } from '@/store';
     6→import { Button } from '@/components/ui/button';
     7→import { Input } from '@/components/ui/input';
     8→import { Badge } from '@/components/ui/badge';
     9→import { Skeleton } from '@/components/ui/skeleton';
    10→import {
    11→  Select,
    12→  SelectContent,
    13→  SelectItem,
    14→  SelectTrigger,
    15→  SelectValue,
    16→} from '@/components/ui/select';
    17→import {
    18→  Table,
    19→  TableBody,
    20→  TableCell,
    21→  TableHead,
    22→  TableHeader,
    23→  TableRow,
    24→} from '@/components/ui/table';
    25→import { toast } from 'sonner';
    26→import {
    27→  ArrowLeft, TrendingUp, Wallet, PiggyBank, TrendingUpIcon,
    28→  ArrowUpCircle, ArrowDownCircle, Loader2,
    29→  Trophy, XCircle, CheckCircle2, Clock, History, Filter,
    30→} from 'lucide-react';
    31→import {
    32→  LineChart,
    33→  Line,
    34→  XAxis,
    35→  YAxis,
    36→  CartesianGrid,
    37→  Tooltip,
    38→  ResponsiveContainer,
    39→  ReferenceLine,
    40→} from 'recharts';
    41→
    42→function formatNaira(amount: number) {
    43→  return `\u20a6${amount.toLocaleString()}`;
    44→}
    45→
    46→interface PricePoint {
    47→  time: string;
    48→  price: number;
    49→  timestamp: number;
    50→}
    51→
    52→interface TradeHistoryItem {
    53→  id: string;
    54→  prediction: 'UP' | 'DOWN';
    55→  amount: number;
    56→  multiplier: number;
    57→  duration: number;
    58→  result: 'win' | 'loss';
    59→  profit: number;
    60→  createdAt: string;
    61→  entryPrice: number;
    62→  exitPrice: number;
    63→}
    64→
    65→type WalletType = 'reward' | 'deposit' | 'profit';
    66→type Prediction = 'UP' | 'DOWN' | null;
    67→
    68→const DURATION_OPTIONS = [
    69→  { label: '10 Seconds', value: 10 },
    70→  { label: '30 Seconds', value: 30 },
    71→  { label: '1 Minute', value: 60 },
    72→  { label: '2 Minutes', value: 120 },
    73→  { label: '5 Minutes', value: 300 },
    74→  { label: '10 Minutes', value: 600 },
    75→];
    76→
    77→const WALLET_CONFIG: Record<WalletType, { label: string; icon: typeof Wallet; color: string; balanceKey: 'reward' | 'deposit' | 'profit' }> = {
    78→  reward: { label: 'Reward Wallet', icon: Wallet, color: 'text-gold', balanceKey: 'reward' },
    79→  deposit: { label: 'Deposit Wallet', icon: PiggyBank, color: 'text-alcoin-blue', balanceKey: 'deposit' },
    80→  profit: { label: 'Profit Wallet', icon: TrendingUpIcon, color: 'text-emerald-400', balanceKey: 'profit' },
    81→};
    82→
    83→const HISTORY_FILTERS = [
    84→  { label: 'All', value: 'all' },
    85→  { label: 'Wins', value: 'win' },
    86→  { label: 'Losses', value: 'loss' },
    87→];
    88→
    89→export default function MarketView() {
    90→  const { wallets, setWallets, setView } = useAppStore();
    91→
    92→  // Chart data
    93→  const [priceData, setPriceData] = useState<PricePoint[]>([]);
    94→  const [currentPrice, setCurrentPrice] = useState<number | null>(null);
    95→  const [prevPrice, setPrevPrice] = useState<number | null>(null);
    96→  const [chartLoading, setChartLoading] = useState(true);
    97→  const priceIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
    98→
    99→  // Trade form
   100→  const [selectedWallet, setSelectedWallet] = useState<WalletType>('deposit');
   101→  const [prediction, setPrediction] = useState<Prediction>(null);
   102→  const [tradeAmount, setTradeAmount] = useState('');
   103→  const [duration, setDuration] = useState<number>(30);
   104→
   105→  // Trade execution
   106→  const [trading, setTrading] = useState(false);
   107→  const [tradeCountdown, setTradeCountdown] = useState<number | null>(null);
   108→  const [tradeResult, setTradeResult] = useState<{ win: boolean; profit: number; totalReturn: number; message: string; multiplier: number } | null>(null);
   109→
   110→  // History
   111→  const [history, setHistory] = useState<TradeHistoryItem[]>([]);
   112→  const [historyLoading, setHistoryLoading] = useState(true);
   113→  const [historyFilter, setHistoryFilter] = useState('all');
   114→
   115→  const fetchPrice = useCallback(async () => {
   116→    try {
   117→      const data = await apiFetch('/api/market/price');
   118→      if (data.price !== undefined) {
   119→        setPrevPrice(currentPrice);
   120→        setCurrentPrice(data.price);
   121→        if (data.history && Array.isArray(data.history)) {
   122→          setPriceData(data.history.slice(-60));
   123→        } else if (data.time) {
   124→          setPriceData((prev) => {
   125→            const next = [...prev, { time: data.time, price: data.price, timestamp: Date.now() }];
   126→            return next.slice(-60);
   127→          });
   128→        }
   129→      }
   130→      setChartLoading(false);
   131→    } catch {
   132→      if (chartLoading) setChartLoading(false);
   133→    }
   134→  }, [currentPrice, chartLoading]);
   135→
   136→  const fetchHistory = useCallback(async () => {
   137→    try {
   138→      setHistoryLoading(true);
   139→      const data = await apiFetch('/api/market/history');
   140→      setHistory(data.trades || []);
   141→    } catch {
   142→      // silent
   143→    } finally {
   144→      setHistoryLoading(false);
   145→    }
   146→  }, []);
   147→
   148→  const refreshWallets = useCallback(async () => {
   149→    try {
   150→      const data = await apiFetch('/api/user/wallets');
   151→      if (data) {
   152→        const getBal = (w: any) => typeof w === 'object' && w !== null ? (w.balance ?? 0) : (w ?? 0);
   153→        setWallets({ reward: getBal(data.reward), deposit: getBal(data.deposit), profit: getBal(data.profit) });
   154→      }
   155→    } catch {
   156→      // silent
   157→    }
   158→  }, [setWallets]);
   159→
   160→  // Initial fetch
   161→  useEffect(() => {
   162→    fetchPrice();
   163→    fetchHistory();
   164→  }, [fetchPrice, fetchHistory]);
   165→
   166→  // Poll price every 2s
   167→  useEffect(() => {
   168→    priceIntervalRef.current = setInterval(fetchPrice, 2000);
   169→    return () => {
   170→      if (priceIntervalRef.current) clearInterval(priceIntervalRef.current);
   171→    };
   172→  }, [fetchPrice]);
   173→
   174→  // Trade countdown
   175→  const countdownRef = useRef<ReturnType<typeof setInterval> | null>(null);
   176→
   177→  useEffect(() => {
   178→    if (tradeCountdown === null) return;
   179→
   180→    countdownRef.current = setInterval(() => {
   181→      setTradeCountdown((prev) => {
   182→        if (prev === null || prev <= 1) {
   183→          if (countdownRef.current) clearInterval(countdownRef.current);
   184→          return 0;
   185→        }
   186→        return prev - 1;
   187→      });
   188→    }, 1000);
   189→
   190→    return () => {
   191→      if (countdownRef.current) clearInterval(countdownRef.current);
   192→    };
   193→  }, [tradeCountdown]);
   194→
   195→  const availableBalance = wallets[WALLET_CONFIG[selectedWallet].balanceKey];
   196→  const amountNum = parseFloat(tradeAmount) || 0;
   197→
   198→  const formatCountdown = (seconds: number) => {
   199→    const m = Math.floor(seconds / 60);
   200→    const s = seconds % 60;
   201→    if (m > 0) return `${m}:${s.toString().padStart(2, '0')}`;
   202→    return `0:${s.toString().padStart(2, '0')}`;
   203→  };
   204→
   205→  const handleTrade = async () => {
   206→    if (!prediction || amountNum <= 0) return;
   207→    if (amountNum > availableBalance) {
   208→      toast.error('Insufficient balance', {
   209→        description: `Your ${WALLET_CONFIG[selectedWallet].label} has ${formatNaira(availableBalance)}`,
   210→      });
   211→      return;
   212→    }
   213→
   214→    setTrading(true);
   215→    setTradeResult(null);
   216→    setTradeCountdown(duration);
   217→
   218→    try {
   219→      const data = await apiFetch('/api/market/trade', {
   220→        method: 'POST',
   221→        body: JSON.stringify({
   222→          wallet: selectedWallet,
   223→          prediction: prediction === 'UP' ? 'buy' : 'sell',
   224→          amount: amountNum,
   225→          duration,
   226→        }),
   227→      });
   228→
   229→      setTradeCountdown(0);
   230→
   231→      // Small delay for dramatic effect
   232→      await new Promise((r) => setTimeout(r, 400));
   233→
   234→      const isWin = data.trade?.result === 'win';
   235→      const multiplierUsed = data.trade?.payout_multiplier || 1.0;
   236→      const totalReturn = isWin ? amountNum * multiplierUsed : 0;
   237→      const profit = isWin ? totalReturn - amountNum : 0;
   238→
   239→      setTradeResult({
   240→        win: isWin,
   241→        profit,
   242→        totalReturn,
   243→        multiplier: multiplierUsed,
   244→        message: data.message || (isWin ? 'Trade successful!' : 'Better luck next time.'),
   245→      });
   246→
   247→      toast.success(
   248→        isWin
   249→          ? `Trade Won! +${formatNaira(profit)}`
   250→          : `Trade Lost`,
   251→        { description: data.message || '' }
   252→      );
   253→
   254→      refreshWallets();
   255→      fetchHistory();
   256→    } catch (err: any) {
   257→      setTradeCountdown(null);
   258→      toast.error(err.message || 'Trade failed');
   259→    } finally {
   260→      setTrading(false);
   261→    }
   262→  };
   263→
   264→  const filteredHistory = historyFilter === 'all'
   265→    ? history
   266→    : history.filter((t) => t.result === historyFilter);
   267→
   268→  // Custom chart tooltip
   269→  const CustomTooltip = ({ active, payload }: any) => {
   270→    if (active && payload && payload.length) {
   271→      return (
   272→        <div className="glass-strong rounded-lg p-2 text-xs border border-white/10">
   273→          <p className="text-muted-foreground">{payload[0].payload.time}</p>
   274→          <p className="font-semibold text-gold">{formatNaira(payload[0].value)}</p>
   275→        </div>
   276→      );
   277→    }
   278→    return null;
   279→  };
   280→
   281→  const priceChange = prevPrice !== null && currentPrice !== null
   282→    ? currentPrice - prevPrice
   283→    : 0;
   284→  const priceUp = priceChange >= 0;
   285→
   286→  return (
   287→    <div className="min-h-screen pb-8">
   288→      {/* Header */}
   289→      <header className="sticky top-0 z-30 glass-strong px-4 py-3 flex items-center gap-3">
   290→        <button
   291→          onClick={() => setView('dashboard')}
   292→          className="w-10 h-10 rounded-xl glass flex items-center justify-center hover:bg-white/10 transition-colors"
   293→        >
   294→          <ArrowLeft className="w-5 h-5" />
   295→        </button>
   296→        <div className="flex items-center gap-2">
   297→          <div className="w-8 h-8 rounded-lg gradient-gold flex items-center justify-center">
   298→            <TrendingUp className="w-4 h-4 text-gold-foreground" />
   299→          </div>
   300→          <h1 className="font-semibold text-lg">AL Coin Market</h1>
   301→        </div>
   302→      </header>
   303→
   304→      <main className="px-4 pt-4 max-w-2xl mx-auto space-y-4">
   305→        {/* Price Chart */}
   306→        <motion.div
   307→          initial={{ opacity: 0, y: 20 }}
   308→          animate={{ opacity: 1, y: 0 }}
   309→          className="glass rounded-xl p-4"
   310→        >
   311→          <div className="flex items-center justify-between mb-3">
   312→            <h2 className="text-sm font-semibold text-muted-foreground">Live Price Chart</h2>
   313→            <div className="flex items-center gap-1.5">
   314→              <div className="w-2 h-2 rounded-full bg-emerald-400 animate-pulse" />
   315→              <span className="text-[10px] text-muted-foreground">LIVE</span>
   316→            </div>
   317→          </div>
   318→
   319→          {chartLoading ? (
   320→            <Skeleton className="h-48 rounded-lg bg-white/5" />
   321→          ) : priceData.length === 0 ? (
   322→            <div className="h-48 rounded-lg bg-white/5 flex items-center justify-center">
   323→              <p className="text-sm text-muted-foreground">Waiting for price data...</p>
   324→            </div>
   325→          ) : (
   326→            <div className="h-48 sm:h-56">
   327→              <ResponsiveContainer width="100%" height="100%">
   328→                <LineChart data={priceData} margin={{ top: 5, right: 5, left: -20, bottom: 0 }}>
   329→                  <CartesianGrid
   330→                    strokeDasharray="3 3"
   331→                    stroke="rgba(212, 175, 55, 0.08)"
   332→                    vertical={false}
   333→                  />
   334→                  <XAxis
   335→                    dataKey="time"
   336→                    tick={{ fontSize: 10, fill: '#8888a0' }}
   337→                    axisLine={{ stroke: 'rgba(255,255,255,0.06)' }}
   338→                    tickLine={false}
   339→                    interval="preserveStartEnd"
   340→                  />
   341→                  <YAxis
   342→                    domain={['auto', 'auto']}
   343→                    tick={{ fontSize: 10, fill: '#8888a0' }}
   344→                    axisLine={{ stroke: 'rgba(255,255,255,0.06)' }}
   345→                    tickLine={false}
   346→                    tickFormatter={(v: number) => `\u20a6${v.toFixed(0)}`}
   347→                  />
   348→                  <Tooltip content={<CustomTooltip />} />
   349→                  <Line
   350→                    type="monotone"
   351→                    dataKey="price"
   352→                    stroke="#d4af37"
   353→                    strokeWidth={2}
   354→                    dot={false}
   355→                    isAnimationActive={false}
   356→                  />
   357→                </LineChart>
   358→              </ResponsiveContainer>
   359→            </div>
   360→          )}
   361→        </motion.div>
   362→
   363→        {/* Current Price Display */}
   364→        <motion.div
   365→          initial={{ opacity: 0, y: 20 }}
   366→          animate={{ opacity: 1, y: 0 }}
   367→          transition={{ delay: 0.05 }}
   368→          className="glass rounded-xl p-4 text-center gold-glow"
   369→        >
   370→          <p className="text-xs text-muted-foreground mb-1">AL Coin Current Price</p>
   371→          <AnimatePresence mode="wait">
   372→            <motion.p
   373→              key={currentPrice}
   374→              initial={{ opacity: 0, y: 5 }}
   375→              animate={{ opacity: 1, y: 0 }}
   376→              exit={{ opacity: 0, y: -5 }}
   377→              className={`text-3xl sm:text-4xl font-bold font-mono ${
   378→                priceUp ? 'text-emerald-400' : priceChange !== 0 ? 'text-red-400' : 'gradient-gold-text'
   379→              }`}
   380→            >
   381→              {currentPrice !== null ? formatNaira(currentPrice) : '---'}
   382→            </motion.p>
   383→          </AnimatePresence>
   384→          {priceChange !== 0 && (
   385→            <p className={`text-xs mt-1 font-medium ${priceUp ? 'text-emerald-400' : 'text-red-400'}`}>
   386→              {priceUp ? '\u25b2' : '\u25bc'} {priceUp ? '+' : ''}{formatNaira(Math.abs(priceChange))}
   387→            </p>
   388→          )}
   389→        </motion.div>
   390→
   391→        {/* ═══════ Place Trade Panel ═══════ */}
   392→        <motion.div
   393→          initial={{ opacity: 0, y: 20 }}
   394→          animate={{ opacity: 1, y: 0 }}
   395→          transition={{ delay: 0.1 }}
   396→          className="glass rounded-xl p-4 space-y-5"
   397→        >
   398→          <h2 className="font-semibold text-sm">Place Trade</h2>
   399→
   400→          {/* 1. Select Wallet — Dropdown */}
   401→          <div className="space-y-2">
   402→            <p className="text-xs text-muted-foreground">Select Wallet</p>
   403→            <Select
   404→              value={selectedWallet}
   405→              onValueChange={(v) => setSelectedWallet(v as WalletType)}
   406→            >
   407→              <SelectTrigger className="w-full bg-white/5 border-white/10 focus:border-gold h-11">
   408→                <div className="flex items-center gap-2">
   409→                  {(() => {
   410→                    const cfg = WALLET_CONFIG[selectedWallet];
   411→                    return <>
   412→                      <cfg.icon className={`w-4 h-4 ${cfg.color}`} />
   413→                      <SelectValue />
   414→                    </>;
   415→                  })()}
   416→                </div>
   417→              </SelectTrigger>
   418→              <SelectContent className="glass-strong border-white/10">
   419→                {(Object.keys(WALLET_CONFIG) as WalletType[]).map((key) => {
   420→                  const cfg = WALLET_CONFIG[key];
   421→                  return (
   422→                    <SelectItem key={key} value={key}>
   423→                      <div className="flex items-center gap-2">
   424→                        <cfg.icon className={`w-4 h-4 ${cfg.color}`} />
   425→                        <span>{cfg.label}</span>
   426→                        <span className="text-muted-foreground ml-auto">({formatNaira(wallets[cfg.balanceKey])})</span>
   427→                      </div>
   428→                    </SelectItem>
   429→                  );
   430→                })}
   431→              </SelectContent>
   432→            </Select>
   433→            <p className="text-[11px] text-muted-foreground">
   434→              Balance: <span className="text-foreground font-medium">{formatNaira(availableBalance)}</span>
   435→            </p>
   436→          </div>
   437→
   438→          {/* 2. Prediction — Buy / Sell */}
   439→          <div className="space-y-2">
   440→            <p className="text-xs text-muted-foreground">Prediction</p>
   441→            <div className="grid grid-cols-2 gap-3">
   442→              <motion.button
   443→                whileTap={{ scale: 0.97 }}
   444→                onClick={() => setPrediction(prediction === 'UP' ? null : 'UP')}
   445→                className={`rounded-xl p-4 flex flex-col items-center gap-2 transition-all border-2 min-h-[80px] ${
   446→                  prediction === 'UP'
   447→                    ? 'bg-emerald-500/15 border-emerald-500/50'
   448→                    : 'glass border-transparent hover:border-emerald-500/20'
   449→                }`}
   450→              >
   451→                <ArrowUpCircle className={`w-8 h-8 ${prediction === 'UP' ? 'text-emerald-400' : 'text-muted-foreground'}`} />
   452→                <span className={`text-sm font-bold ${prediction === 'UP' ? 'text-emerald-400' : 'text-muted-foreground'}`}>
   453→                  Buy
   454→                </span>
   455→                <span className={`text-[10px] ${prediction === 'UP' ? 'text-emerald-400/70' : 'text-muted-foreground/60'}`}>
   456→                  Chart goes UP
   457→                </span>
   458→              </motion.button>
   459→
   460→              <motion.button
   461→                whileTap={{ scale: 0.97 }}
   462→                onClick={() => setPrediction(prediction === 'DOWN' ? null : 'DOWN')}
   463→                className={`rounded-xl p-4 flex flex-col items-center gap-2 transition-all border-2 min-h-[80px] ${
   464→                  prediction === 'DOWN'
   465→                    ? 'bg-red-500/15 border-red-500/50'
   466→                    : 'glass border-transparent hover:border-red-500/20'
   467→                }`}
   468→              >
   469→                <ArrowDownCircle className={`w-8 h-8 ${prediction === 'DOWN' ? 'text-red-400' : 'text-muted-foreground'}`} />
   470→                <span className={`text-sm font-bold ${prediction === 'DOWN' ? 'text-red-400' : 'text-muted-foreground'}`}>
   471→                  Sell
   472→                </span>
   473→                <span className={`text-[10px] ${prediction === 'DOWN' ? 'text-red-400/70' : 'text-muted-foreground/60'}`}>
   474→                  Chart goes DOWN
   475→                </span>
   476→              </motion.button>
   477→            </div>
   478→          </div>
   479→
   480→          {/* 3. Investment Amount */}
   481→          <div className="space-y-2">
   482→            <p className="text-xs text-muted-foreground">Investment</p>
   483→            <div className="relative">
   484→              <span className="absolute left-3 top-1/2 -translate-y-1/2 text-sm text-muted-foreground font-medium">
   485→                \u20a6
   486→              </span>
   487→              <Input
   488→                type="number"
   489→                inputMode="decimal"
   490→                placeholder="0"
   491→                value={tradeAmount}
   492→                onChange={(e) => setTradeAmount(e.target.value)}
   493→                className="pl-8 pr-16 bg-white/5 border-white/10 focus:border-gold h-12 text-lg font-semibold"
   494→                min="0"
   495→                max={availableBalance}
   496→              />
   497→              <button
   498→                type="button"
   499→                onClick={() => setTradeAmount(availableBalance.toString())}
   500→                className="absolute right-2 top-1/2 -translate-y-1/2 px-2 py-1 rounded text-[10px] font-medium text-gold bg-gold/10 hover:bg-gold/20 transition-colors"
   501→              >
   502→                MAX
   503→              </button>
   504→            </div>
   505→            {amountNum > availableBalance && amountNum > 0 && (
   506→              <p className="text-[11px] text-red-400">
   507→                Amount exceeds your {WALLET_CONFIG[selectedWallet].label} balance of {formatNaira(availableBalance)}
   508→              </p>
   509→            )}
   510→          </div>
   511→
   512→          {/* 4. Trade Duration */}
   513→          <div className="space-y-2">
   514→            <p className="text-xs text-muted-foreground">Trade Duration</p>
   515→            <Select
   516→              value={duration.toString()}
   517→              onValueChange={(v) => setDuration(parseInt(v))}
   518→            >
   519→              <SelectTrigger className="w-full bg-white/5 border-white/10 focus:border-gold h-11">
   520→                <Clock className="w-4 h-4 text-muted-foreground mr-2" />
   521→                <SelectValue />
   522→              </SelectTrigger>
   523→              <SelectContent className="glass-strong border-white/10">
   524→                {DURATION_OPTIONS.map((opt) => (
   525→                  <SelectItem key={opt.value} value={opt.value.toString()}>
   526→                    {opt.label}
   527→                  </SelectItem>
   528→                ))}
   529→              </SelectContent>
   530→            </Select>
   531→          </div>
   532→
   533→          {/* Place Trade Button */}
   534→          <Button
   535→            onClick={handleTrade}
   536→            disabled={!prediction || amountNum <= 0 || amountNum > availableBalance || trading}
   537→            className="w-full gradient-gold text-gold-foreground font-bold h-13 text-base disabled:opacity-40 disabled:cursor-not-allowed"
   538→          >
   539→            {!prediction || amountNum <= 0 ? (
   540→              'Select Prediction & Amount'
   541→            ) : amountNum > availableBalance ? (
   542→              'Insufficient Balance'
   543→            ) : trading ? (
   544→              <span className="flex items-center gap-2">
   545→                <Loader2 className="w-5 h-5 animate-spin" />
   546→                Trading...
   547→              </span>
