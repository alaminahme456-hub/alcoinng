    91→// Local error boundary — prevents app-level crash from showing error.tsx
    92→class MarketErrorBoundary extends Component<
    93→  { children: ReactNode },
    94→  { hasError: boolean; error: Error | null }
    95→> {
    96→  constructor(props: { children: ReactNode }) {
    97→    super(props);
    98→    this.state = { hasError: false, error: null };
    99→  }
   100→  static getDerivedStateFromError(error: Error) {
   101→    return { hasError: true, error };
   102→  }
   103→  componentDidCatch(error: Error, info: ErrorInfo) {
   104→    console.error('MarketView error boundary caught:', error, info);
   105→  }
   106→  render() {
   107→    if (this.state.hasError) {
   108→      return (
   109→        <div className="min-h-screen flex flex-col items-center justify-center bg-background text-foreground p-4">
   110→          <div className="max-w-md w-full text-center space-y-4">
   111→            <div className="w-16 h-16 rounded-full bg-destructive/10 flex items-center justify-center mx-auto">
   112→              <XCircle className="w-8 h-8 text-destructive" />
   113→            </div>
   114→            <h2 className="text-lg font-bold">Trade Error</h2>
   115→            <p className="text-sm text-muted-foreground">
   116→              {this.state.error?.message || 'Something went wrong with the market.'}
   117→            </p>
   118→            <Button
   119→              onClick={() => { this.setState({ hasError: false, error: null }); }}
   120→              className="gradient-gold text-gold-foreground font-semibold"
   121→            >
   122→              Try Again
   123→            </Button>
   124→          </div>
   125→        </div>
   126→      );
   127→    }
   128→    return this.props.children;
   129→  }
   130→}

   315→  const countdownDisplayRef = useRef(0);

   468→      monitorInfoRef.current = null;

   488→      );
   489→    }
   490→    return null;
   491→  };

   531→          {chartLoading ? (

   534→            <div className="h-48 rounded-lg bg-white/5 flex items-center justify-center">

   537→          ) : (

   543→                      <stop offset="0%" stopColor="#34d399" stopOpacity={0.25} />
