    89→    89→export default function MarketView() {
    90→    90→  const { wallets, setWallets, setView } = useAppStore();
    91→    91→
    92→    92→  // Chart data
    93→    93→  const [priceData, setPriceData] = useState<PricePoint[]>([]);
    94→    94→  const [currentPrice, setCurrentPrice] = useState<number | null>(null);
    95→    95→  const [prevPrice, setPrevPrice] = useState<number | null>(null);
    96→    96→  const [chartLoading, setChartLoading] = useState(true);
    97→    97→  const priceIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
    98→    98→
    99→    99→  // Trade form
   100→   100→  const [selectedWallet, setSelectedWallet] = useState<WalletType>('deposit');
   101→   101→  const [prediction, setPrediction] = useState<Prediction>(null);
   102→   102→  const [tradeAmount, setTradeAmount] = useState('');
   103→   103→  const [duration, setDuration] = useState<number>(30);
   104→   104→
   105→   105→  // Trade execution
   106→   106→  const [trading, setTrading] = useState(false);
   107→   107→  const [tradeCountdown, setTradeCountdown] = useState<number | null>(null);
   108→   108→  const [tradeResult, setTradeResult] = useState<{ win: boolean; profit: number; totalReturn: number; message: string; multiplier: number } | null>(null);
   109→   109→
   110→   110→  // History
   111→   111→  const [history, setHistory] = useState<TradeHistoryItem[]>([]);
   112→   112→  const [historyLoading, setHistoryLoading] = useState(true);
   113→   113→  const [historyFilter, setHistoryFilter] = useState('all');
   114→   114→
   115→   115→  const fetchPrice = useCallback(async () => {
   116→   116→    try {
   117→   117→      const data = await apiFetch('/api/market/price');
   118→   118→      if (data.price !== undefined) {
   119→   119→        setPrevPrice(currentPrice);
   120→   120→        setCurrentPrice(data.price);
   121→   121→        if (data.history && Array.isArray(data.history)) {
   122→   122→          setPriceData(data.history.slice(-60));
   123→   123→        } else if (data.time) {
   124→   124→          setPriceData((prev) => {
   125→   125→            const next = [...prev, { time: data.time, price: data.price, timestamp: Date.now() }];
   126→   126→            return next.slice(-60);
   127→   127→          });
   128→   128→        }
   129→   129→      }
   130→   130→      setChartLoading(false);
   131→   131→    } catch {
   132→   132→      if (chartLoading) setChartLoading(false);
   133→   133→    }
   134→   134→  }, [currentPrice, chartLoading]);
   135→   135→
   136→   136→  const fetchHistory = useCallback(async () => {
   137→   137→    try {
   138→   138→      setHistoryLoading(true);
   139→   139→      const data = await apiFetch('/api/market/history');
   140→   140→      setHistory(data.trades || []);
   141→   141→    } catch {
   142→   142→      // silent
   143→   143→    } finally {
   144→   144→      setHistoryLoading(false);
   145→   145→    }
   146→   146→  }, []);
   147→   147→
   148→   148→  const refreshWallets = useCallback(async () => {
   149→   149→    try {
   150→   150→      const data = await apiFetch('/api/user/wallets');
   151→   151→      if (data) {
   152→   152→        const getBal = (w: any) => typeof w === 'object' && w !== null ? (w.balance ?? 0) : (w ?? 0);
   153→   153→        setWallets({ reward: getBal(data.reward), deposit: getBal(data.deposit), profit: getBal(data.profit) });
   154→   154→      }
   155→   155→    } catch {
   156→   156→      // silent
   157→   157→    }
   158→   158→  }, [setWallets]);
   159→   159→
   160→   160→  // Initial fetch
   161→   161→  useEffect(() => {
   162→   162→    fetchPrice();
   163→   163→    fetchHistory();
   164→   164→  }, [fetchPrice, fetchHistory]);
   165→   165→
   166→   166→  // Poll price every 2s
   167→   167→  useEffect(() => {
   168→   168→    priceIntervalRef.current = setInterval(fetchPrice, 2000);
   169→   169→    return () => {
   170→   170→      if (priceIntervalRef.current) clearInterval(priceIntervalRef.current);
   171→   171→    };
   172→   172→  }, [fetchPrice]);
   173→   173→
   174→   174→  // Trade countdown
   175→   175→  const countdownRef = useRef<ReturnType<typeof setInterval> | null>(null);
   176→   176→
   177→   177→  useEffect(() => {
   178→   178→    if (tradeCountdown === null) return;
   179→   179→
   180→   180→    countdownRef.current = setInterval(() => {
   181→   181→      setTradeCountdown((prev) => {
   182→   182→        if (prev === null || prev <= 1) {
   183→   183→          if (countdownRef.current) clearInterval(countdownRef.current);
   184→   184→          return 0;
   185→   185→        }
   186→   186→        return prev - 1;
   187→   187→      });
   188→   188→    }, 1000);
   189→   189→
   190→   190→    return () => {
   191→   191→      if (countdownRef.current) clearInterval(countdownRef.current);
   192→   192→    };
   193→   193→  }, [tradeCountdown]);
   194→   194→
   195→   195→  const availableBalance = wallets[WALLET_CONFIG[selectedWallet].balanceKey];
   196→   196→  const amountNum = parseFloat(tradeAmount) || 0;
   197→   197→
   198→   198→  const formatCountdown = (seconds: number) => {
   199→   199→    const m = Math.floor(seconds / 60);
   200→   200→    const s = seconds % 60;
   201→   201→    if (m > 0) return `${m}:${s.toString().padStart(2, '0')}`;
   202→   202→    return `0:${s.toString().padStart(2, '0')}`;
   203→   203→  };
   204→   204→
   205→   205→  const handleTrade = async () => {
   206→   206→    if (!prediction || amountNum <= 0) return;
   207→   207→    if (amountNum > availableBalance) {
   208→   208→      toast.error('Insufficient balance', {
   209→   209→        description: `Your ${WALLET_CONFIG[selectedWallet].label} has ${formatNaira(availableBalance)}`,
   210→   210→      });
   211→   211→      return;
   212→   212→    }
   213→   213→
   214→   214→    setTrading(true);
   215→   215→    setTradeResult(null);
   216→   216→    setTradeCountdown(duration);
   217→   217→
   218→   218→    try {
   219→   219→      const data = await apiFetch('/api/market/trade', {
   220→   220→        method: 'POST',
   221→   221→        body: JSON.stringify({
   222→   222→          wallet: selectedWallet,
   223→   223→          prediction: prediction === 'UP' ? 'buy' : 'sell',
   224→   224→          amount: amountNum,
   225→   225→          duration,
   226→   226→        }),
   227→   227→      });
   228→   228→
   229→   229→      setTradeCountdown(0);
   230→   230→
   231→   231→      // Small delay for dramatic effect
   232→   232→      await new Promise((r) => setTimeout(r, 400));
   233→   233→
   234→   234→      const isWin = data.trade?.result === 'win';
   235→   235→      const multiplierUsed = data.trade?.payout_multiplier || 1.0;
   236→   236→      const totalReturn = isWin ? amountNum * multiplierUsed : 0;
   237→   237→      const profit = isWin ? totalReturn - amountNum : 0;
   238→   238→
   239→   239→      setTradeResult({
   240→   240→        win: isWin,
   241→   241→        profit,
   242→   242→        totalReturn,
   243→   243→        multiplier: multiplierUsed,
   244→   244→        message: data.message || (isWin ? 'Trade successful!' : 'Better luck next time.'),
   245→   245→      });
   246→   246→
   247→   247→      toast.success(
   248→   248→        isWin
   249→   249→          ? `Trade Won! +${formatNaira(profit)}`
   250→   250→          : `Trade Lost`,
   251→   251→        { description: data.message || '' }
   252→   252→      );
   253→   253→
   254→   254→      refreshWallets();
   255→   255→      fetchHistory();
   256→   256→    } catch (err: any) {
   257→   257→      setTradeCountdown(null);
   258→   258→      toast.error(err.message || 'Trade failed');
   259→   259→    } finally {
   260→   260→      setTrading(false);
   261→   261→    }
   262→   262→  };
   263→   263→
   264→   264→  const filteredHistory = historyFilter === 'all'
   265→   265→    ? history
   266→   266→    : history.filter((t) => t.result === historyFilter);
   267→   267→
   268→   268→  // Custom chart tooltip
   269→   269→  const CustomTooltip = ({ active, payload }: any) => {

   271→   271→      return (
   272→   272→        <div className="glass-strong rounded-lg p-2 text-xs border border-white/10">
   273→   273→          <p className="text-muted-foreground">{payload[0].payload.time}</p>
   274→   274→          <p className="font-semibold text-gold">{formatNaira(payload[0].value)}</p>
   275→   275→        </div>
   276→   276→      );
   277→   277→    }
   278→   278→    return null;
   279→   279→  };
   280→   280→
   281→   281→  const priceChange = prevPrice !== null && currentPrice !== null
   282→   282→    ? currentPrice - prevPrice
   283→   283→    : 0;
   284→   284→  const priceUp = priceChange >= 0;
   285→   285→
   286→   286→  return (
   287→   287→    <div className="min-h-screen pb-8">
   288→   288→      {/* Header */}
   289→   289→      <header className="sticky top-0 z-30 glass-strong px-4 py-3 flex items-center gap-3">
   290→   290→        <button
   291→   291→          onClick={() => setView('dashboard')}
   292→   292→          className="w-10 h-10 rounded-xl glass flex items-center justify-center hover:bg-white/10 transition-colors"
   293→   293→        >
   294→   294→          <ArrowLeft className="w-5 h-5" />
   295→   295→        </button>
   296→   296→        <div className="flex items-center gap-2">
   297→   297→          <div className="w-8 h-8 rounded-lg gradient-gold flex items-center justify-center">
   298→   298→            <TrendingUp className="w-4 h-4 text-gold-foreground" />
   299→   299→          </div>
   300→   300→          <h1 className="font-semibold text-lg">AL Coin Market</h1>
   301→   301→        </div>
   302→   302→      </header>
   303→   303→
   304→   304→      <main className="px-4 pt-4 max-w-2xl mx-auto space-y-4">
   305→   305→        {/* Price Chart */}
   306→   306→        <motion.div
   307→   307→          initial={{ opacity: 0, y: 20 }}
   308→   308→          animate={{ opacity: 1, y: 0 }}
   309→   309→          className="glass rounded-xl p-4"
   310→   310→        >
   311→   311→          <div className="flex items-center justify-between mb-3">
   312→   312→            <h2 className="text-sm font-semibold text-muted-foreground">Live Price Chart</h2>
   313→   313→            <div className="flex items-center gap-1.5">
   314→   314→              <div className="w-2 h-2 rounded-full bg-emerald-400 animate-pulse" />
   315→   315→              <span className="text-[10px] text-muted-foreground">LIVE</span>
   316→   316→            </div>
   317→   317→          </div>
   318→   318→
   319→   319→          {chartLoading ? (
   320→   320→            <Skeleton className="h-48 rounded-lg bg-white/5" />
   321→   321→          ) : priceData.length === 0 ? (
   322→   322→            <div className="h-48 rounded-lg bg-white/5 flex items-center justify-center">
   323→   323→              <p className="text-sm text-muted-foreground">Waiting for price data...</p>
   324→   324→            </div>
   325→   325→          ) : (
   326→   326→            <div className="h-48 sm:h-56">
   327→   327→              <ResponsiveContainer width="100%" height="100%">
   328→   328→                <LineChart data={priceData} margin={{ top: 5, right: 5, left: -20, bottom: 0 }}>
   329→   329→                  <CartesianGrid
   330→   330→                    strokeDasharray="3 3"
   331→   331→                    stroke="rgba(212, 175, 55, 0.08)"
   332→   332→                    vertical={false}
   333→   333→                  />
   334→   334→                  <XAxis
   335→   335→                    dataKey="time"
   336→   336→                    tick={{ fontSize: 10, fill: '#8888a0' }}
   337→   337→                    axisLine={{ stroke: 'rgba(255,255,255,0.06)' }}
   338→   338→                    tickLine={false}
   339→   339→                    interval="preserveStartEnd"
   340→   340→                  />
   341→   341→                  <YAxis
   342→   342→                    domain={['auto', 'auto']}
   343→   343→                    tick={{ fontSize: 10, fill: '#8888a0' }}
   344→   344→                    axisLine={{ stroke: 'rgba(255,255,255,0.06)' }}
   345→   345→                    tickLine={false}
   346→   346→                    tickFormatter={(v: number) => `\u20a6${v.toFixed(0)}`}
   347→   347→                  />
   348→   348→                  <Tooltip content={<CustomTooltip />} />
   349→   349→                  <Line
   350→   350→                    type="monotone"
   351→   351→                    dataKey="price"
   352→   352→                    stroke="#d4af37"
   353→   353→                    strokeWidth={2}
   354→   354→                    dot={false}
   355→   355→                    isAnimationActive={false}
   356→   356→                  />
   357→   357→                </LineChart>
   358→   358→              </ResponsiveContainer>
   359→   359→            </div>
   360→   360→          )}
   361→   361→        </motion.div>
   362→   362→
   363→   363→        {/* Current Price Display */}
   364→   364→        <motion.div
   365→   365→          initial={{ opacity: 0, y: 20 }}
   366→   366→          animate={{ opacity: 1, y: 0 }}
   367→   367→          transition={{ delay: 0.05 }}
   368→   368→          className="glass rounded-xl p-4 text-center gold-glow"
   369→   369→        >
   370→   370→          <p className="text-xs text-muted-foreground mb-1">AL Coin Current Price</p>
   371→   371→          <AnimatePresence mode="wait">
   372→   372→            <motion.p

   374→   374→              initial={{ opacity: 0, y: 5 }}
   375→   375→              animate={{ opacity: 1, y: 0 }}

   377→   377→              className={`text-3xl sm:text-4xl font-bold font-mono ${
   378→   378→                priceUp ? 'text-emerald-400' : priceChange !== 0 ? 'text-red-400' : 'gradient-gold-text'
   379→   379→              }`}

   384→   384→          {priceChange !== 0 && (

   386→   386→              {priceUp ? '\u25b2' : '\u25bc'} {priceUp ? '+' : ''}{formatNaira(Math.abs(priceChange))}
   387→   387→            </p>
   388→   388→          )}
   389→   389→        </motion.div>
   390→   390→
   391→   391→        {/* ═══════ Place Trade Panel ═══════ */}
   392→   392→        <motion.div
   393→   393→          initial={{ opacity: 0, y: 20 }}
   394→   394→          animate={{ opacity: 1, y: 0 }}
   395→   395→          transition={{ delay: 0.1 }}
   396→   396→          className="glass rounded-xl p-4 space-y-5"
   397→   397→        >
   398→   398→          <h2 className="font-semibold text-sm">Place Trade</h2>
   399→   399→
   400→   400→          {/* 1. Select Wallet — Dropdown */}
   401→   401→          <div className="space-y-2">
   402→   402→            <p className="text-xs text-muted-foreground">Select Wallet</p>
   403→   403→            <Select
   404→   404→              value={selectedWallet}
   405→   405→              onValueChange={(v) => setSelectedWallet(v as WalletType)}

   407→   407→              <SelectTrigger className="w-full bg-white/5 border-white/10 focus:border-gold h-11">

   409→   409→                  {(() => {
   410→   410→                    const cfg = WALLET_CONFIG[selectedWallet];
   411→   411→                    return <>
   412→   412→                      <cfg.icon className={`w-4 h-4 ${cfg.color}`} />
   413→   413→                      <SelectValue />

   415→   415→                  })()}
   416→   416→                </div>
   417→   417→              </SelectTrigger>
   418→   418→              <SelectContent className="glass-strong border-white/10">
   419→   419→                {(Object.keys(WALLET_CONFIG) as WalletType[]).map((key) => {

   421→   421→                  return (
   422→   422→                    <SelectItem key={key} value={key}>
   423→   423→                      <div className="flex items-center gap-2">
   424→   424→                        <cfg.icon className={`w-4 h-4 ${cfg.color}`} />

   431→   431→              </SelectContent>

   438→   438→          {/* 2. Prediction — Buy / Sell */}

   441→   441→            <div className="grid grid-cols-2 gap-3">
   442→   442→              <motion.button
   443→   443→                whileTap={{ scale: 0.97 }}
   444→   444→                onClick={() => setPrediction(prediction === 'UP' ? null : 'UP')}
   445→   445→                className={`rounded-xl p-4 flex flex-col items-center gap-2 transition-all border-2 min-h-[80px] ${
   446→   446→                  prediction === 'UP'

   450→   450→              >

   452→   452→                <span className={`text-sm font-bold ${prediction === 'UP' ? 'text-emerald-400' : 'text-muted-foreground'}`}>
   453→   453→                  Buy
   454→   454→                </span>
   455→   455→                <span className={`text-[10px] ${prediction === 'UP' ? 'text-emerald-400/70' : 'text-muted-foreground/60'}`}>
   456→   456→                  Chart goes UP
   457→   457→                </span>
   458→   458→              </motion.button>
   459→   459→
   460→   460→              <motion.button
